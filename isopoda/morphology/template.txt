function makeLayer(group){
 const wrap=document.createElement('div');wrap.className=`exploded-part ${group.className}`;
 wrap.dataset.part=group.name;wrap.dataset.semantic=group.semantic;
 const canvas=document.createElement('canvas');canvas.width=canvas.height=64;
 const ctx=canvas.getContext('2d');ctx.imageSmoothingEnabled=false;
 for(const module of anatomy){
  if(!group.match(module.region))continue;
  for(const [x,y,color] of module.cells){
   if(!color)continue;if(group.side==='top'&&y>=0)continue;if(group.side==='bottom'&&y<0)continue;
   const px=x+32,py=y+32;if(px<0||px>=64||py<0||py>=64)continue;
   ctx.fillStyle=color;ctx.fillRect(px,py,1,1);
  }
 }
 wrap.append(canvas);stage.append(wrap);return wrap;
}

function drawModules(canvas,modules,{filter=()=>true,scale=1,offsetX=0,offsetY=0}={}){
 const ctx=canvas.getContext('2d');ctx.clearRect(0,0,64,64);ctx.imageSmoothingEnabled=false;
 for(const module of modules){
  if(!filter(module.region))continue;
  for(const [x,y,color] of module.cells){
   const px=Math.round(32+offsetX+x*scale),py=Math.round(32+offsetY+y*scale);
   if(px<0||px>=64||py<0||py>=64)continue;
   ctx.fillStyle=color;ctx.fillRect(px,py,Math.max(1,Math.round(scale)),Math.max(1,Math.round(scale)));
  }
 }
}

}

function templateFor(key){
 const v=model.visual;
 if(key==='body')return {template:v.morphologyKey||'body-proportion layer',confidence:'render morphology'};
 if(key==='cephalon')return {template:v.cephalon.template||'head template',confidence:v.cephalon.confidence||'render proxy'};
 if(key==='antennae')return {template:v.antennae.template||'antenna template',confidence:v.antennae.confidence||'render proxy'};
 if(key==='pereon'||key==='epimera')return {template:v.pereon.template||'pereon template',confidence:v.pereon.confidence||'render proxy'};
 if(key==='surface')return {template:v.surface.template||v.surface.sculpture||'surface template',confidence:v.surface.confidence||'render proxy'};
 if(key==='pereopods')return {template:v.legs.template||'leg eco-morph',confidence:v.legs.confidence||'eco-morph proxy'};
 if(key==='pleon')return {template:v.pleotelson.template||'posterior body template',confidence:v.pleotelson.confidence||'render proxy'};
 if(key==='pleotelson')return {template:v.pleotelson.shape||v.pleotelson.template||'tail template',confidence:v.pleotelson.confidence||'render proxy'};
 if(key==='uropods')return {template:`${v.uropods.mode||'uropod'} / ${v.uropods.template||'tail template'}`,confidence:v.uropods.confidence||'render proxy'};
 if(key==='conglobation')return {template:v.conglobation.strategy||v.conglobation.ability||'conglobation',confidence:v.conglobation.confidence||'body-plan proxy'};
 return {template:'—',confidence:'—'};
}

function hitTest(ev){
 for(let i=layers.length-1;i>=0;i--){
  const layer=layers[i],canvas=layer.querySelector('canvas'),rect=canvas.getBoundingClientRect();
  if(ev.clientX<rect.left||ev.clientX>=rect.right||ev.clientY<rect.top||ev.clientY>=rect.bottom)continue;
  const x=Math.floor((ev.clientX-rect.left)/rect.width*64),y=Math.floor((ev.clientY-rect.top)/rect.height*64);
  if(x<0||x>63||y<0||y>63)continue;
  if(canvas.getContext('2d').getImageData(x,y,1,1).data[3]>0)return layer.dataset.semantic;
 }
 return null;
}
